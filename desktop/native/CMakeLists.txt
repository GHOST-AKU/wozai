cmake_minimum_required(VERSION 3.20)
project(wozai_bluetooth LANGUAGES CXX)

if(NOT WIN32)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Bluetooth is supported on Windows and Linux.")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
    find_path(JNI_INCLUDE jni.h PATHS "$ENV{JAVA_HOME}/include" NO_DEFAULT_PATH REQUIRED)
    find_path(JNI_PLATFORM_INCLUDE jni_md.h PATHS "$ENV{JAVA_HOME}/include/linux" NO_DEFAULT_PATH REQUIRED)
    add_library(wozai_bluetooth SHARED linux_bluetooth.cpp)
    target_compile_features(wozai_bluetooth PRIVATE cxx_std_17)
    target_include_directories(wozai_bluetooth PRIVATE "${JNI_INCLUDE}" "${JNI_PLATFORM_INCLUDE}")
    target_link_libraries(wozai_bluetooth PRIVATE PkgConfig::GIO)
    target_compile_options(wozai_bluetooth PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(nearbyim_launcher linux_launcher.cpp)
    target_compile_features(nearbyim_launcher PRIVATE cxx_std_17)
    target_compile_options(nearbyim_launcher PRIVATE -Wall -Wextra -Wpedantic -Werror)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(linux_bluetooth_tests linux_bluetooth_tests.cpp)
        target_compile_features(linux_bluetooth_tests PRIVATE cxx_std_17)
        target_include_directories(linux_bluetooth_tests PRIVATE "${JNI_INCLUDE}" "${JNI_PLATFORM_INCLUDE}")
        target_link_libraries(linux_bluetooth_tests PRIVATE PkgConfig::GIO)
        target_compile_options(linux_bluetooth_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        add_test(NAME linux_bluez_contract COMMAND dbus-run-session -- $<TARGET_FILE:linux_bluetooth_tests>)
        set_tests_properties(linux_bluez_contract PROPERTIES TIMEOUT 30)
    endif()
    return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the Bluetooth DLL for x64 to match the packaged JDK.")
endif()

# Only JNI headers are needed; this DLL must not link against jvm.dll.
if(NOT DEFINED ENV{JAVA_HOME})
    message(FATAL_ERROR "Set JAVA_HOME to the x64 JDK used to build the desktop application.")
endif()
find_path(JNI_INCLUDE jni.h PATHS "$ENV{JAVA_HOME}/include" NO_DEFAULT_PATH REQUIRED)
find_path(JNI_PLATFORM_INCLUDE jni_md.h PATHS "$ENV{JAVA_HOME}/include/win32" NO_DEFAULT_PATH REQUIRED)

add_library(wozai_bluetooth SHARED windows_bluetooth.cpp)
target_compile_features(wozai_bluetooth PRIVATE cxx_std_17)
target_include_directories(wozai_bluetooth PRIVATE "${JNI_INCLUDE}" "${JNI_PLATFORM_INCLUDE}")
target_compile_definitions(wozai_bluetooth PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_link_libraries(wozai_bluetooth PRIVATE Ws2_32 Bthprops)
if(MSVC)
    target_compile_options(wozai_bluetooth PRIVATE /W4 /permissive- /EHsc)
    # Shipping the static runtime avoids requiring a VC redistributable installer.
    set_property(TARGET wozai_bluetooth PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(wozai_bluetooth PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(bluetooth_native_tests socket_tests.cpp)
    target_compile_features(bluetooth_native_tests PRIVATE cxx_std_17)
    target_include_directories(bluetooth_native_tests PRIVATE "${JNI_INCLUDE}" "${JNI_PLATFORM_INCLUDE}")
    target_compile_definitions(bluetooth_native_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
    target_link_libraries(bluetooth_native_tests PRIVATE Ws2_32 Bthprops)
    if(MSVC)
        target_compile_options(bluetooth_native_tests PRIVATE /W4 /permissive- /EHsc)
        set_property(TARGET bluetooth_native_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME bluetooth_native_lifecycle COMMAND bluetooth_native_tests)
    set_tests_properties(bluetooth_native_lifecycle PROPERTIES TIMEOUT 20)
endif()
